import matplotlib.pyplot as plt

# Cargar el conjunto de datos Iris
iris = load_iris()
X = iris.data  # Las características
y = iris.target  # Las etiquetas de las especies
species = iris.target_names  # Nombres de las especies

# Crear una figura 3D
fig = plt.figure(figsize=(10, 7))

for i, color in enumerate(colors):
    # Filtrar las muestras para cada especie
    ax.scatter(X[y == i, 0], X[y == i, 1], X[y == i, 2], color=color, label=species[i], s=50)

# Etiquetas de los ejes
ax.set_xlabel('Sepal Length (cm)')
ax.set_ylabel('Sepal Width (cm)')
ax.set_zlabel('Petal Length (cm)')
ax.legend()
ax.set_title("3D Scatter plot of Iris Dataset")

import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt
from sklearn.metrics import confusion_matrix

def plot_confusion_matrix(y_true, y_pred, class_names, title='Matriz de Confusión', xlabel='Predicciones', ylabel='Etiquetas Reales'):
    """
    Grafica una matriz de confusión.

    :param y_true: Array de etiquetas reales
    :param y_pred: Array de etiquetas predichas
    :param class_names: Lista con los nombres de las clases
    :param title: Título del gráfico
    :param xlabel: Etiqueta del eje X
    :param ylabel: Etiqueta del eje Y
    """
    # Calcula la matriz de confusión
    cm = confusion_matrix(y_true, y_pred)
    
    # Configura la figura
    plt.figure(figsize=(4, 4))
    
    # Crea un heatmap para la matriz de confusión
    sns.heatmap(cm, annot=True, fmt='d', cmap='Blues', 
                xticklabels=class_names, yticklabels=class_names, cbar=False)
    
    # Añade títulos y etiquetas
    plt.title(title)
    plt.xlabel(xlabel)
    plt.ylabel(ylabel)

    # Muestra la gráfica
    plt.show()

# Parámetros
n_splits = 5  # Número de repeticiones
skf = StratifiedKFold(n_splits=2)  # Validación cruzada 2-fold

# Listas para almacenar diferencias de precisión
differences = []

# Realizar el 5x2 cross-validation
for _ in range(n_splits):
    for train_idx, test_idx in skf.split(X, y):
        # Dividir los datos en entrenamiento y prueba para cada fold
        X_train, X_test = X[train_idx], X[test_idx]
        y_train, y_test = y[train_idx], y[test_idx]

        # Entrenar y predecir con Naïve Bayes
        model_nb.fit(X_train, y_train)
        y_pred_nb = model_nb.predict(X_test)
        acc_nb = accuracy_score(y_test, y_pred_nb)

        # Entrenar y predecir con KNN
        model_knn.fit(X_train, y_train)
        y_pred_knn = model_knn.predict(X_test)
        acc_knn = accuracy_score(y_test, y_pred_knn)

        # Guardar la diferencia de precisión
        differences.append(acc_nb - acc_knn)

# Calcular la media y la varianza de las diferencias
mean_diff = np.mean(differences)
var_diff = np.var(differences, ddof=1)

# Calcular el estadístico t
t_stat = mean_diff / np.sqrt(var_diff / len(differences))
p_value_t_test = 2 * (1 - stats.t.cdf(abs(t_stat), df=len(differences) - 1))

# Mostrar los resultados del 5x2 cross-validation t-test
print(f"Estadístico t: {t_stat}")
print(f"P-valor del t-test 5x2: {p_value_t_test}")
